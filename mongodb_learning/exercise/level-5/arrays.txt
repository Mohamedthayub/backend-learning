Level 5: Array Exercises (20 Exercises)

Use the skills array.

Examples:

Students with JavaScript.
db.students.find({skills:'JavaScript'})
---------------------------------------
Students with Node.js.
db.students.find({skills:'Node.js'})
--------------------------------------
Students with both JavaScript and MongoDB.
db.students.find({$and:[{skills:'Javascript'},{skills:'MongoDB'}]})
-------------------------------------------------------
Add React to one student.
db.students.updateOne({name:'Gokul'},{$push:{skills:"Next.js"}})
----------------------------------------------------------------
Add Express to everyone.
db.students.updateMany({},{$push:{skills:'Express'}})
----------------------------------------------------
Remove Python.
db.students.updateOne({name:'Gokul'},{$pull:{skills:'Python'}})
---------------------------------------------------------------------
Remove the last skill.
db.students.updateOne({name:'Gokul'},{$pop:{skills:1}})
-------------------------------------------------------
Remove the first skill.
db.students.updateOne({name:'Gokul'},{$pop:{skills:-1}})
--------------------------------------
Replace one skill with another.
db.students.updateOne({name:'Hari'},{$set:{'skills.0':'HTML'}})
---------------------------------------------------------
Prevent duplicate skills.
db.students.updateOne({name:'Lakshmi'},{$addToSet:{skills:'Thayub'}})
-------------------------------------
update the array data 
db.students.updateOne(
  { name: "Gokul" },
  {
    $set: {
      "skills.2": "MongoDB"
    }
  }
)
-----------------------------------
